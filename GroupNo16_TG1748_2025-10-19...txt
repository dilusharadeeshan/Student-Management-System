1. Medical Apply Attendance
----------------------------



DROP VIEW IF EXISTS medicalApplyAttendance;

CREATE VIEW medicalApplyAttendance AS 
SELECT attendance_id,reg_no,course_code,session_date,session_type,session_hours,status 
FROM attendance 
WHERE status='Medical';

SELECT * FROM medicalApplyAttendance;

3. Attendance data with medical approved (attendance table)
-----------------------------------------------------------



DROP VIEW IF EXISTS attendance_with_medical_approval;


CREATE VIEW attendance_with_medical_approval AS
SELECT 
    a.attendance_id,
    a.reg_no,
    a.course_code,
    a.session_date,
    a.session_type,
    a.session_hours,
    CASE 
        WHEN a.status = 'Medical' AND m.Approved=TRUE THEN 'Present/M'
        WHEN a.status = 'Medical' AND m.Approved=FALSE THEN 'Absent/M'
        ELSE a.status
    END AS status
FROM attendance a
LEFT JOIN Medical_record m ON 
    a.Medical_id = m.Medical_id;


SELECT * FROM attendance_with_medical_approval;


DROP VIEW IF EXISTS attendance_with_medical_approval;

4.Attendance percentage without considering medical
---------------------------------------------------

DROP VIEW IF EXISTS attendance_percentage_without_medical;

CREATE VIEW attendance_percentage_without_medical AS
SELECT 
    reg_no,
    course_code,
    session_type,
    SUM(CASE WHEN status = 'Present' THEN session_hours ELSE 0 END) AS attended_hours,
    SUM(session_hours) AS total_hours,
    SUM(CASE WHEN status = 'Present' THEN 1 ELSE 0 END) AS attended_sessions,
    COUNT(*) AS total_sessions,
    
    ROUND(
        (SUM(CASE WHEN status = 'Present' THEN session_hours ELSE 0 END) / 
        SUM(session_hours)
    ) * 100, 2
    ) AS attendance_percentage
FROM attendance
GROUP BY reg_no, course_code,session_type
ORDER BY reg_no, course_code,session_type;

SELECT * FROM attendance_percentage_without_medical;

DROP VIEW IF EXISTS attendance_without_medical;

--------------------------------------------------------------------

    session_type,
    SUM(CASE WHEN status = 'Present' THEN session_hours ELSE 0 END) AS attended_hours,
    SUM(session_hours) AS total_hours,

        (SUM(CASE WHEN status = 'Present' THEN session_hours ELSE 0 END) /

    ) * 100, 2
    ) AS attendance_percentage,

	WHEN

            (SUM(CASE WHEN status = 'Present' THEN session_hours ELSE 0 END) /
