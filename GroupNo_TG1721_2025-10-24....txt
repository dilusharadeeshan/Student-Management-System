DROP VIEW IF EXISTS vw_CA_pass_fail;
CREATE VIEW vw_CA_pass_fail AS
SELECT 
    Reg_no,
    Course_code,
    CA_marks,
    CASE 
        WHEN Course_code IN ('ICT1222', 'ICT1233', 'ICT1253', 'TCS1212', 'TMS1233') THEN 
            CASE WHEN CA_marks >= 20 THEN 'Pass' ELSE 'Fail' END
        WHEN Course_code IN ('ENG1222', 'ICT1212', 'ICT1242') THEN 
            CASE WHEN CA_marks >= 15 THEN 'Pass' ELSE 'Fail' END
        ELSE 'Unknown'
    END AS CA_Status
FROM (
    -- ENG1222 (30% CA - threshold: 15)
    SELECT 
        Reg_no,
        Course_code,
        `Total_CA_marks(30%)` AS CA_marks
    FROM vw_ENG1222_marks
    
    UNION ALL
    
    -- ICT1212 (30% CA - threshold: 15)
    SELECT 
        Reg_no,
        Course_code,
        `Total_CA_marks(30%)` AS CA_marks
    FROM vw_ICT1212_marks
    
    UNION ALL
    
    -- ICT1222 (40% CA - threshold: 20)
    SELECT 
        Reg_no,
        Course_code,
        `Total_CA_marks(40%)` AS CA_marks
    FROM vw_ICT1222_marks
    
    UNION ALL
    
    -- ICT1233 (40% CA - threshold: 20)
    SELECT 
        Reg_no,
        Course_code,
        `Total_CA_marks(40%)` AS CA_marks
    FROM vw_ICT1233_marks
    
    UNION ALL
    
    -- ICT1242 (30% CA - threshold: 15)
    SELECT 
        Reg_no,
        Course_code,
        `Total_CA_marks(30%)` AS CA_marks
    FROM vw_ICT1242_marks
    
    UNION ALL
    
    -- ICT1253 (40% CA - threshold: 20)
    SELECT 
        Reg_no,
        Course_code,
        `Total_CA_marks(40%)` AS CA_marks
    FROM vw_ICT1253_marks
    
    UNION ALL
    
    -- TCS1212 (40% CA - threshold: 20)
    SELECT 
        Reg_no,
        Course_code,
        `Total_CA_marks(40%)` AS CA_marks
    FROM vw_TCS1212_marks
    
    UNION ALL
    
    -- TMS1233 (40% CA - threshold: 20)
    SELECT 
        Reg_no,
        Course_code,
        `Total_CA_marks(40%)` AS CA_marks
    FROM vw_TMS1233_marks
) AS all_ca_marks
ORDER BY reg_no, course_code;

SELECT * FROM vw_CA_pass_fail;
